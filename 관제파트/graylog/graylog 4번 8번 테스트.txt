$udp = New-Object System.Net.Sockets.UdpClient

try {
    for ($i = 1; $i -le 12; $i++) {
        $log = @{
            version = "1.1"
            host = "security-test"
            short_message = "Test login failure $i"
            level = 4
            _rule = "login-bruteforce"
            _src_ip = "203.0.113.30"
            _username = "test_admin"
        } | ConvertTo-Json -Compress

        $bytes = [System.Text.Encoding]::UTF8.GetBytes($log)

        [void]$udp.Send(
            $bytes,
            $bytes.Length,
            "127.0.0.1",
            12201
        )

        Start-Sleep -Milliseconds 200
    }
}
finally {
    $udp.Close()
}

Write-Host "테스트 로그인 실패 로그 12건 전송 완료"
