
# Graylog ED-07 Wazuh Rule 테스트
# GELF UDP Input: localhost:12201

$GRAYLOG_HOST = "127.0.0.1"
$GRAYLOG_PORT = 12201

# 테스트할 Rule ID
$RULE_ID = "100222"

$rules = @{
    "100211" = @{
        level = 12
        description = "Repeated login failures detected"
        srcip = "203.0.113.70"
        file = ""
        event = ""
    }
    "100220" = @{
        level = 15
        description = "Web shell suspected"
        srcip = ""
        file = "/var/www/html/test-shell.php"
        event = "added"
    }
    "100221" = @{
        level = 8
        description = "Web file modification detected"
        srcip = ""
        file = "/var/www/html/index.html"
        event = "modified"
    }
    "100222" = @{
        level = 10
        description = "Executable file deletion detected"
        srcip = ""
        file = "/var/www/html/test.exe"
        event = "deleted"
    }
}

$r = $rules[$RULE_ID]

if (-not $r) {
    throw "등록되지 않은 Rule ID: $RULE_ID"
}

$udp = New-Object System.Net.Sockets.UdpClient

try {
    $log = @{
        version = "1.1"
        host = "wazuh-test-agent"
        short_message = "Wazuh simulation rule $RULE_ID"
        level = 4

        _wazuh_rule_id = $RULE_ID
        _wazuh_rule_level = $r.level
        _wazuh_rule_description = $r.description
        _wazuh_agent_name = "web-server-01"
        _wazuh_data_srcip = $r.srcip
        _wazuh_file = $r.file
        _wazuh_syscheck_event = $r.event

        _test_source = "wazuh-ed07-lab"
    } | ConvertTo-Json -Compress

    $bytes = [System.Text.Encoding]::UTF8.GetBytes($log)

    [void]$udp.Send(
        $bytes,
        $bytes.Length,
        $GRAYLOG_HOST,
        $GRAYLOG_PORT
    )

    Write-Host "Wazuh 모의 경보 전송 완료"
    Write-Host "Rule ID: $RULE_ID"
    Write-Host "Level: $($r.level)"
    Write-Host "IP: $($r.srcip)"
}
finally {
    $udp.Close()
}
