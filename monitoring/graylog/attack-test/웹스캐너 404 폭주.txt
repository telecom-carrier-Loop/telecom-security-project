
# ED-09 웹스캐너 404 폭주 탐지 테스트
# Graylog GELF UDP 12201 포트 사용

$GRAYLOG_HOST = "127.0.0.1"
$GRAYLOG_PORT = 12201

$TEST_IP = "203.0.113.40"
$LOG_COUNT = 35

$udp = New-Object System.Net.Sockets.UdpClient

try {
    for ($i = 1; $i -le $LOG_COUNT; $i++) {

        $log = @{
            version = "1.1"
            host = "web-security-test"
            short_message = "Simulated HTTP 404 scan request #$i"
            level = 4

            _rule = "web-scan"
            _src_ip = $TEST_IP
            _status_code = 404
            _http_method = "GET"
            _request_path = "/test-missing-$i"
            _scanner = "security-lab"
        } | ConvertTo-Json -Compress

        $bytes = [System.Text.Encoding]::UTF8.GetBytes($log)

        [void]$udp.Send(
            $bytes,
            $bytes.Length,
            $GRAYLOG_HOST,
            $GRAYLOG_PORT
        )

        Write-Host "[$i/$LOG_COUNT] HTTP 404 테스트 로그 전송"

        Start-Sleep -Milliseconds 100
    }
}
finally {
    $udp.Close()
}

Write-Host ""
Write-Host "ED-09 테스트 로그 35건 전송 완료"
