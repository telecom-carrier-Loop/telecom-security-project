    $log = @{
      version = '1.1'
      host = 'telecom-test-pc2'
      short_message = "[SIMULATED] ____ event $i/$Count"
      level = 4
    _rule = 'portscan'
    _src_ip = '198.51.100.41'
    _test_source = 'TELECOM-MIGRATION-20261026'
    _simulation = 'true'
    _test_case = '다중공격'
    _test_origin = 'graylog-gelf'
    _username = 'test_admin'
    }
    $log._test_seq = $i
    $json = $log | ConvertTo-Json -Compress -Depth 8
    $bytes = [Text.Encoding]::UTF8.GetBytes($json)
    [void]$udp.Send($bytes, $bytes.Length, $GraylogIP, $UdpPort)
    Start-Sleep -Milliseconds 170
  }
